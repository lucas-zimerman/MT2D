cmake_minimum_required(VERSION 3.12)
project(desert CXX)
include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/MT2D.cmake)
if(NOT MT2D_CSCRIPT_PARSER)
    message(STATUS "MT2D example 'desert' loads text scripts and needs MT2D_CSCRIPT_PARSER, skipped")
    return()
endif()
file(GLOB DESERT_ASSETS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} assets/*.png assets/*.ogg assets/*.wav scripts/*.txt)
mt2d_add_example(desert SOURCES src/main.cpp SDL ASSETS ${DESERT_ASSETS} DOS_NAME desert)
